import java.util.*;

public class burger {
    public static void main(String args[]) {

        Scanner sc = new Scanner(System.in);

        String str = sc.nextLine();

        int piece[] = new int[3];
        int price[] = new int[3];

        int r = sc.nextInt();

        for (int i = 0; i < 3; i++) {
            piece[i] = sc.nextInt();
            price[i] = sc.nextInt();
        }

        int b = 0, s = 0, c = 0;

        for (char ch : str.toCharArray()) {
            if (ch == 'B')
                b++;
            else if (ch == 'S')
                s++;
            else
                c++;
        }

        int count = 0;

        while (true) {

            
            int buyB = Math.max(0, b - piece[0]);
            int buyS = Math.max(0, s - piece[1]);
            int buyC = Math.max(0, c - piece[2]);

            
            int cost = buyB * price[0]
                     + buyS * price[1]
                     + buyC * price[2];

           
            if (cost > r)
                break;

            
            r -= cost;

           
            piece[0] -= b;
            piece[1] -= s;
            piece[2] -= c;

            
            piece[0] += buyB;
            piece[1] += buyS;
            piece[2] += buyC;

            count++;
        }

        System.out.println(count);
    }
}
