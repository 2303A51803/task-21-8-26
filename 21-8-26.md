1)select e.emp\_id, e.emp\_name, d.dept\_name, d.location from employ as e 

&#x20;inner join department as d on e.dept\_id = d.dept\_id;

2)select e.emp\_id, e.emp\_name, d.dept\_name from employ as e 

&#x20;left join department as d on e.dept\_id = d.dept\_id;

3)select d.dept\_id, d.dept\_name from department as d

&#x20;left join employ as e on d.dept\_id = e.dept\_id where e.emp\_id is null;

4)select e.emp\_name, p.project\_name, d.dept\_name, ep.hours from employ as e

&#x20;join employee\_project as ep on e.emp\_id = ep.emp\_id 

&#x20;join project as p on ep.project\_id = p.project\_id 

&#x20;join department as d on p.dept\_id = d.dept\_id;

5)select p.project\_id, p.project\_name, e.emp\_name from project as p 

&#x20;left join employee\_project as ep on p.project\_id = ep.project\_id 

&#x20;left join employ as e on ep.emp\_id = e.emp\_id;

6)select e.emp\_name , m.emp\_name as manager from employ as e 

&#x20;left join employ as m on e.manager\_id = m.emp\_id;
